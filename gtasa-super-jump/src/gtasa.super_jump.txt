// ============================================================================
//  gtasa.super_jump  -  GTA San Andreas (Android) CLEO super jump   [v2]
// ----------------------------------------------------------------------------
//  Target : GTA SA Mobile + CLEO Android 2.0.x
//  Format : .csi  (menu-invoked; slide top->bottom through centre to open menu)
//  Build  : Sanny Builder 4.2.0, edit mode sa_mobile
//
//  v2 replaces the v1 trigger. v1 relied on 00E1 (pad button state), which on
//  Android is NOT driven by the on-screen touch HUD on every build, so the jump
//  button appeared dead. v2 triggers on the CLEO Android touch API (0DE0), which
//  the runtime itself implements, and keeps 00E1 as an extra trigger.
//
//  Touch regions (0DE0) = 3x3 grid of the screen:
//      1 LeftTop   2 LeftCenter   3 LeftBottom
//      4 CenterTop 5 Center       6 CenterBottom
//      7 RightTop  8 RightCenter  9 RightBottom   <- jump button lives here
//  See docs/ANDROID_NOTES.md: regions are cells, not buttons, so a press
//  anywhere in the bottom-right cell counts. That is the deliberate trade-off
//  for a trigger that actually fires on device.
// ============================================================================
{$CLEO .csi}

// ------------------------------ configuration -------------------------------
// 20@  key trigger id A      14 = SQUARE "Jump / Zoom In" (on foot)
// 26@  key trigger id B      16 = CROSS "Sprint"      (extra safety net)
// 21@  upward launch speed   18.0 m/s (vanilla jump ~7)
// 22@  touch region A         9 = RightBottom (jump button corner)
// 23@  touch region B         8 = RightCenter
// 25@  cooldown frames       20 (~0.3 s) after each jump
// Set 23@ to 0 to disable region B, or set 22@/23@ to 0 to disable touch
// regions entirely (key triggers only).
// ----------------------------------------------------------------------------
:SUPERJUMP_INIT
0006: 20@ = 14
0006: 26@ = 16
0007: 21@ = 18.0
0006: 22@ = 9
0006: 23@ = 8
0006: 25@ = 20
0006: 1@ = 0
0006: 4@ = 0
0001: wait 500 ms

// -------------------------------- main loop ---------------------------------
:SUPERJUMP_MAIN
0001: wait 0 ms
0006: 0@ = 0

// cooldown after a jump: skip input, keep the loop responsive
00D6: if 0
0019:   4@ > 0
004D: jump_if_false @SUPERJUMP_INPUT
000A: 4@ += -1
0002: jump @SUPERJUMP_MAIN

// --- trigger 1: touch region A (bottom-right cell = jump button area) --------
:SUPERJUMP_INPUT
00D6: if 0
0039:   22@ == 0
004D: jump_if_false @SUPERJUMP_REGION_A
0002: jump @SUPERJUMP_REGION_OFF

:SUPERJUMP_REGION_A
00D6: if 0
0DE0: 3@ = get_touch_point_state 22@ mintime 0
004D: jump_if_false @SUPERJUMP_REGION_B
0006: 0@ = 1
0002: jump @SUPERJUMP_CHECK_KEY

// --- trigger 2: touch region B (right-centre cell, alternative layouts) -----
:SUPERJUMP_REGION_B
00D6: if 0
0039:   23@ == 0
004D: jump_if_false @SUPERJUMP_REGION_B_DO
0002: jump @SUPERJUMP_REGION_OFF

:SUPERJUMP_REGION_B_DO
00D6: if 0
0DE0: 3@ = get_touch_point_state 23@ mintime 0
004D: jump_if_false @SUPERJUMP_REGION_OFF
0006: 0@ = 1
0002: jump @SUPERJUMP_CHECK_KEY

:SUPERJUMP_REGION_OFF
0006: 0@ = 0

// --- trigger 3: jump action bit (some builds/gamepads do drive this) --------
:SUPERJUMP_CHECK_KEY
00D6: if 0
0039:   0@ == 1
004D: jump_if_false @SUPERJUMP_KEY_A
0002: jump @SUPERJUMP_EDGE

:SUPERJUMP_KEY_A
00D6: if 0
00E1:   player 0 pressed_key 20@
004D: jump_if_false @SUPERJUMP_KEY_B
0006: 0@ = 1
0002: jump @SUPERJUMP_EDGE

:SUPERJUMP_KEY_B
00D6: if 0
00E1:   player 0 pressed_key 26@
004D: jump_if_false @SUPERJUMP_RELEASED
0006: 0@ = 1
0002: jump @SUPERJUMP_EDGE

// nothing pressed: re-arm the latch so the next press fires again
:SUPERJUMP_RELEASED
0006: 1@ = 0
0002: jump @SUPERJUMP_MAIN

// --------------------------- press latch (edge) -----------------------------
:SUPERJUMP_EDGE
00D6: if 0
0039:   1@ == 1
004D: jump_if_false @SUPERJUMP_DO
0002: jump @SUPERJUMP_MAIN

// ------------------------------- super jump --------------------------------
:SUPERJUMP_DO
0006: 1@ = 1
0006: 4@ = 25
01F5: 2@ = get_player_actor 0

// guard: no player actor this frame -> abort cleanly, never crash the thread
00D6: if 0
0039:   2@ == 0
004D: jump_if_false @SUPERJUMP_LAUNCH
0002: jump @SUPERJUMP_MAIN

:SUPERJUMP_LAUNCH
// 1) the game's own jump task
05BC: AS_actor 2@ jump 1
// 2) raw upward velocity
083C: set_actor 2@ velocity_in_direction_XYZ 0.0 0.0 21@
// 3) hard lift so the jump is visible even if velocity is overridden
00A0: store_actor 2@ position_to 5@ 6@ 7@
000B: 7@ += 1.5
00A1: set_actor 2@ coords 5@ 6@ 7@
0002: jump @SUPERJUMP_MAIN
