// ============================================================================
//  gtasa.robber  -  GTA San Andreas (Android) CLEO robbery mod   [v3]
// ----------------------------------------------------------------------------
//  Target : GTA SA Mobile + CLEO Android 2.0.x
//  Format : .csi (menu-invoked)
//
//  HOW IT WORKS: aim a gun at a pedestrian for a moment. He throws his hands up,
//  you take his cash, and he runs away. There is no button to press - the mod
//  reacts to your crosshair, so it cannot be defeated by touch detection.
//
//  v3 change: the touch-region opcode 0DE0 was removed. On builds whose CLEO
//  runtime lacks it, calling it kills the script instantly - which is exactly
//  what "+$1 then nothing" looked like. v3 uses only core SA opcodes.
//
//  SELF-CHECK, so you always know what is happening:
//    + $1 the moment it starts        -> the script loaded
//    + $1 every 5 seconds afterwards  -> the main loop is alive
//    + $50..$250 while aiming at a ped -> a robbery happened
// ============================================================================
{$CLEO .csi}

:ROBBER_INIT
0006: 3@ = 0
0006: 4@ = 0
0001: wait 500 ms

// proof of life
0109: player 0 money += 1

// -------------------------------- main loop ---------------------------------
:ROBBER_MAIN
0001: wait 250 ms

// heartbeat: +$1 every 20 ticks (~5 s) proves the loop is running
000A: 3@ += 1
00D6: if 0
0019:   3@ > 19
004D: jump_if_false @ROBBER_COOLDOWN
0109: player 0 money += 1
0006: 3@ = 0

// cooldown between robberies (40 ticks ~ 10 s)
:ROBBER_COOLDOWN
00D6: if 0
0019:   4@ > 0
004D: jump_if_false @ROBBER_AIM
000A: 4@ += -1
0002: jump @ROBBER_MAIN

// --- is CJ aiming at somebody? ---------------------------------------------
:ROBBER_AIM
00D6: if 0
0457: 6@ = player 0 aiming_at_actor
004D: jump_if_false @ROBBER_MAIN

// no target (0) -> keep waiting
00D6: if 0
0039:   6@ == 0
004D: jump_if_false @ROBBER_ROB
0002: jump @ROBBER_MAIN

// -------------------------------- the robbery -------------------------------
:ROBBER_ROB
01F5: 2@ = get_player_actor 0
0209: 5@ = random_int_in_ranges 50 250

0109: player 0 money += 5@

// the victim reacts: hands up, then runs off
05C4: AS_actor 6@ hands_up 4000
05DB: AS_actor 6@ flee_from_actor 2@ from_origin_radius 25 timelimit 9000

0006: 4@ = 40
0002: jump @ROBBER_MAIN
